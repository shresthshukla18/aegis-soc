cmake_minimum_required(VERSION 3.20)
project(AegisSOC VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(AEGIS_ENABLE_PCAP "Enable live packet capture through libpcap/Npcap" ON)
add_executable(aegis_soc src/main.cpp src/engine.cpp src/http_server.cpp src/packet_capture.cpp)
target_include_directories(aegis_soc PRIVATE include)
target_compile_definitions(aegis_soc PRIVATE AEGIS_VERSION="${PROJECT_VERSION}" AEGIS_WEB_DIR="${CMAKE_SOURCE_DIR}/web" AEGIS_CONFIG_DIR="${CMAKE_SOURCE_DIR}/config")
if(MSVC)
  target_compile_options(aegis_soc PRIVATE /W4 /permissive-)
  target_link_libraries(aegis_soc PRIVATE ws2_32)
else()
  target_compile_options(aegis_soc PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(aegis_soc PRIVATE pthread)
endif()
if(AEGIS_ENABLE_PCAP)
  find_path(PCAP_INCLUDE_DIR pcap/pcap.h)
  if(NOT PCAP_INCLUDE_DIR)
    find_path(PCAP_INCLUDE_DIR pcap.h)
  endif()
  find_library(PCAP_LIBRARY NAMES pcap wpcap)
  if(PCAP_INCLUDE_DIR AND PCAP_LIBRARY)
    target_include_directories(aegis_soc PRIVATE ${PCAP_INCLUDE_DIR})
    target_link_libraries(aegis_soc PRIVATE ${PCAP_LIBRARY})
    target_compile_definitions(aegis_soc PRIVATE AEGIS_HAS_PCAP=1)
    message(STATUS "PCAP support enabled")
  else()
    message(STATUS "PCAP not found: simulation mode remains available")
  endif()
endif()
install(TARGETS aegis_soc RUNTIME DESTINATION bin)
